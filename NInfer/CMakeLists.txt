cmake_minimum_required(VERSION 3.28)

option(NINFER_WIN7 "Build the Win7 x64 / sm75 text-inference baseline" OFF)
option(NINFER_TEXT_ONLY "Exclude image/video decoding and media acquisition dependencies" ${NINFER_WIN7})
option(NINFER_DISABLE_NVTX "Build without NVTX instrumentation" ${NINFER_WIN7})

# real emits a cubin without a PTX fallback requiring a newer driver's JIT.
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
  if(NINFER_WIN7)
    set(CMAKE_CUDA_ARCHITECTURES 75-real CACHE STRING "CUDA architectures to build")
  else()
    set(CMAKE_CUDA_ARCHITECTURES 75 CACHE STRING "CUDA architectures to build")
  endif()
endif()
if(NOT CMAKE_CUDA_ARCHITECTURES MATCHES "^(75(-real)?|86|120a)$")
  message(FATAL_ERROR "Unsupported CUDA architecture: ${CMAKE_CUDA_ARCHITECTURES}")
endif()
string(REGEX REPLACE "-real$" "" NINFER_CUDA_ARCH "${CMAKE_CUDA_ARCHITECTURES}")
if(NINFER_WIN7)
  if(NOT CMAKE_CUDA_ARCHITECTURES STREQUAL "75-real")
    message(FATAL_ERROR "The Win7 baseline requires CMAKE_CUDA_ARCHITECTURES=75-real")
  endif()
  set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
  set(CMAKE_CUDA_RUNTIME_LIBRARY Static)
endif()

project(ninfer-2080ti-22g LANGUAGES C CXX CUDA)

option(NINFER_BUILD_APPS "Build the NInfer CLI" ON)
set(NINFER_SERVER_DEFAULT ON)
if(NINFER_WIN7)
  set(NINFER_SERVER_DEFAULT OFF)
endif()
option(NINFER_BUILD_SERVER "Build the HTTP server in addition to the CLI" ${NINFER_SERVER_DEFAULT})
option(BUILD_TESTING "Build NInfer tests" OFF)
option(NINFER_BUILD_BENCHMARKS "Build NInfer benchmarks" OFF)
if(NINFER_WIN7 AND NINFER_BUILD_SERVER)
  message(FATAL_ERROR "The Win7 baseline does not build the HTTP server")
endif()

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(CMAKE_CUDA_COMPILER_VERSION VERSION_LESS 12.0)
  set(CMAKE_CUDA_STANDARD 17)
else()
  set(CMAKE_CUDA_STANDARD 20)
endif()
set(CMAKE_CUDA_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

if(CMAKE_GENERATOR MATCHES "Ninja")
  set_property(GLOBAL PROPERTY JOB_POOLS cuda_link=1)
  set(CMAKE_JOB_POOL_LINK cuda_link)
endif()

if(CMAKE_CUDA_COMPILER_VERSION VERSION_LESS 12.8)
  if(CMAKE_CUDA_COMPILER_VERSION VERSION_LESS 11.7 OR NOT NINFER_CUDA_ARCH STREQUAL "75")
    message(FATAL_ERROR "CUDA 11.7 through 12.7 is supported only by the sm75 compatibility path")
  endif()
endif()

if(NINFER_WIN7)
  if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The Win7 baseline requires a 64-bit Windows MSVC build host")
  endif()
  if(NOT MSVC_VERSION EQUAL 1929)
    message(FATAL_ERROR "The Win7 baseline requires MSVC v142 14.29, not the newest VS toolset")
  endif()
  if(CMAKE_CUDA_COMPILER_VERSION VERSION_LESS 11.7 OR
     NOT CMAKE_CUDA_COMPILER_VERSION VERSION_LESS 12.0)
    message(FATAL_ERROR "The Win7 baseline requires CUDA 11.7 or 11.8 (11.7 is the qualification target)")
  endif()
  if(NOT NINFER_TEXT_ONLY OR NOT NINFER_DISABLE_NVTX)
    message(FATAL_ERROR "The Win7 baseline requires NINFER_TEXT_ONLY and NINFER_DISABLE_NVTX")
  endif()
  if(BUILD_TESTING OR NINFER_BUILD_BENCHMARKS)
    message(FATAL_ERROR "Use tests/win7 for baseline host tests; the full upstream GPU suite is not Win7-qualified")
  endif()
  set(NINFER_WIN7_THUNKS
      "${CMAKE_CURRENT_SOURCE_DIR}/../third_party/YY-Thunks/objs/x64/YY_Thunks_for_Win7.obj"
      CACHE FILEPATH "Win7 x64 YY-Thunks object from the enclosing nerv repository")
  if(NOT EXISTS "${NINFER_WIN7_THUNKS}")
    message(FATAL_ERROR "Missing Win7 compatibility object: ${NINFER_WIN7_THUNKS}")
  endif()
  add_compile_definitions(NINFER_WIN7=1 _WIN32_WINNT=0x0601 WINVER=0x0601
                          WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
  add_compile_options(
    "$<$<COMPILE_LANGUAGE:C,CXX>:/utf-8>"
    "$<$<COMPILE_LANGUAGE:CXX>:/Zc:__cplusplus>"
    "$<$<COMPILE_LANGUAGE:CUDA>:-Xcompiler=/utf-8,/Zc:__cplusplus>")
endif()

if(NINFER_CUDA_ARCH STREQUAL "120a")
  add_compile_definitions(NINFER_SM120A=1)
else()
  add_compile_definitions(NINFER_SM${NINFER_CUDA_ARCH}=1)
endif()
if(NINFER_TEXT_ONLY)
  add_compile_definitions(NINFER_TEXT_ONLY=1)
endif()
if(NINFER_DISABLE_NVTX)
  add_compile_definitions(NINFER_DISABLE_NVTX=1)
endif()

set(NINFER_BUILD_MEDIA_ACQUIRE OFF)
if(NINFER_BUILD_APPS OR BUILD_TESTING)
  set(NINFER_BUILD_MEDIA_ACQUIRE ON)
endif()
set(NINFER_BUILD_PROMPT_INPUT ${NINFER_BUILD_APPS})
set(NINFER_BUILD_SERVE OFF)
if((NINFER_BUILD_APPS AND NINFER_BUILD_SERVER) OR BUILD_TESTING)
  set(NINFER_BUILD_SERVE ON)
endif()

find_package(CUDAToolkit REQUIRED)
find_package(Threads REQUIRED)
set(NINFER_CUDART_TARGET CUDA::cudart)
if(NINFER_WIN7)
  set(NINFER_CUDART_TARGET CUDA::cudart_static)
endif()
if(NOT NINFER_TEXT_ONLY)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(FFMPEG REQUIRED IMPORTED_TARGET
    libavformat>=60 libavcodec>=60 libavutil>=58 libswscale>=7)
  if(NINFER_BUILD_MEDIA_ACQUIRE)
    pkg_check_modules(LIBCURL REQUIRED IMPORTED_TARGET libcurl>=7.85)
  endif()
endif()

add_subdirectory(src)
if(NINFER_BUILD_APPS)
  add_subdirectory(apps)
endif()
if(BUILD_TESTING)
  enable_testing()
  add_subdirectory(tests)
endif()
if(NINFER_BUILD_BENCHMARKS)
  add_subdirectory(bench)
endif()
